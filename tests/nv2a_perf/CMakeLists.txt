cmake_minimum_required(VERSION 3.20)
project(nv2a_perf_test C)
# The time is the test's, and so are the samples: no thread looks on its own.
set(CMAKE_C_STANDARD 11)
add_executable(nv2a-perf-test nv2a_perf_test.c ../../src/kernel/nv2a_perf.c)
target_include_directories(nv2a-perf-test PRIVATE ../../src/kernel ../../src)
target_compile_definitions(nv2a-perf-test PRIVATE NV2A_PERF_NO_SAMPLER)
if(MSVC)
    target_compile_options(nv2a-perf-test PRIVATE /W4)
    target_compile_definitions(nv2a-perf-test PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
enable_testing()
add_test(NAME nv2a-perf COMMAND nv2a-perf-test)
add_test(NAME nv2a-perf-log COMMAND nv2a-perf-test log)
set_tests_properties(nv2a-perf-log PROPERTIES ENVIRONMENT RECOMP_PERF_LOG=100)
